const move = <T>(arr: T[], i: number, d: number): T[] | null => {
  const j = i + d;
  if (j < 0 || j >= arr.length) return null;
  const a = [...arr];
  [a[i], a[j]] = [a[j], a[i]];
  return a;
};

        uid += 1;
        set({ toast: { id: uid, title, sub, icon, action } });
        clearTimeout(toastTimer);
        // longer messages (sign-in problems, server errors) stay up long enough to read
        toastTimer = setTimeout(() => set({ toast: null }), Math.min(6000, Math.max(2600, 1000 + sub.length * 45)));
      };

      /**
       * A server-backed change: show it at once, send it, and put the touched state back with the server's
       * reason if it's refused. Catalog writes then refresh the catalog so its version catches up.
       */
      const write = async <T, K extends keyof State>(opts: {
        keys: K[];
        apply?: (s: State) => Partial<State>;
        call: (r: Remote) => Promise<T>;
        done?: (res: T, s: State) => Partial<State> | void;
        fail: string;
        catalog?: boolean;
      }): Promise<T | null> => {
        const before = get();
        const snapshot = Object.fromEntries(opts.keys.map((k) => [k, before[k]])) as Partial<State>;
        if (opts.apply) set(opts.apply(before));
        try {
          const res = await opts.call(server());
          const patch = opts.done?.(res, get());
          if (patch) set(patch);
          if (opts.catalog) server().catalogChanged();
          return res;
        } catch (e) {
          set(snapshot);
          toast(opts.fail, remote?.errorMessage(e) ?? 'Try again in a moment.', 'error');
          return null;
        }
      };

      const replaceOrder = (o: Order) =>
        set((s) => ({ orders: s.orders.some((x) => x.id === o.id) ? s.orders.map((x) => (x.id === o.id ? o : x)) : [o, ...s.orders] }));
      const orderByNo = (no: string) => get().orders.find((o) => o.no === no);

      return {

          const hasKitchenFees = kitchenIds.some((id) => s.brands.find((brand) => brand.id === id)?.deliveryFee != null);
          const deliveryFee = hasKitchenFees ? kitchenIds.reduce((sum, id) => sum + (s.brands.find((brand) => brand.id === id)?.deliveryFee ?? s.fee), 0) : s.fee;
          return computeTotals({ cart: s.cart, dishOf: s.dishOf, mode: s.mode, speed: s.speed, fee: deliveryFee, freeAt: s.freeAt, promo: s.promo });
        },

        showToast: toast,
        hideToast: () => set({ toast: null }),

              catalogEtag: etag,
              menu: byOrder(c.dishes).map(({ order: _o, ...d }) => d),
              cats: byOrder(c.categories).map(({ id, name, on, img }) => ({ id, name, on, img })),
              brands: byOrder(c.brands).map(({ order: _o, ...b }) => ({ ...b, icon: b.icon as IconName })),
              banners: byOrder(c.banners).map(({ order: _o, ...b }) => b),
              secs: c.sections.map((x) => ({ ...x, icon: x.icon as IconName })),

        const p = (persisted ?? {}) as Partial<State>;
        if (version < 2) {
          const { cart, saved, addrs, addr, cards, defCard, inbox, prefs, theme, onboarded, orderNotif, mode, speed, payM } = p;
          return { cart, saved, addrs, addr, cards, defCard, inbox, prefs: { ...prefs, lang: 'العربية' }, theme, onboarded, orderNotif, mode, speed, payM } as AppStore;
        }
        if (version < 3) return { ...p, prefs: { ...p.prefs, lang: 'العربية' } } as AppStore;
        return p as AppStore;
      },
    },
  ),
);

  /** Applies a patch to one document and bumps the catalog version in the same batch. */
  async function patch<T extends z.ZodType>(ref: DocumentReference, what: string, data: Record<string, unknown>, schema: T, idField = 'id') {
    await existing(ref, what);
    const batch = db.batch();
    batch.update(ref, { ...data, updatedAt: FieldValue.serverTimestamp() });
    bumpCatalog(db, batch);
    await batch.commit();
    const snap = await ref.get();
    return schema.parse({ [idField]: snap.id, ...snap.data() }) as z.infer<T>;
  }

  async function reorder(name: string, ids: string[]) {
    if (new Set(ids).size !== ids.length) throw badRequest('Each item can appear once.');
    const batch = db.batch();

import type { AdminCatalog, Booking, Brand, Category, Deal, Dish, IssueResult, LikeResult, Me, Order, Post, Promo, Review, StoreSettings } from '@fig/shared';

import { registerRemote, type Remote } from '@/store/app-store';

import { api, errorMessage } from './api';
import { catalogChanged } from './sync';

const enc = encodeURIComponent;

/** The store's server calls, over the Fig & Fennel API. */
export const remote: Remote = {
  validatePromo: (code) => api.post<{ code: string; pct: number }>('/v1/promos/validate', { code }),
  redeem: (id) => api.post<Me>(`/v1/me/rewards/${enc(id)}/redeem`),
  rate: (id, input) => api.post<Order>(`/v1/orders/${enc(id)}/rating`, input),
  reportIssue: (orderId, type, text) =>
    orderId ? api.post<IssueResult>(`/v1/orders/${enc(orderId)}/issues`, { type, text }) : api.post<IssueResult>('/v1/issues', { type, text, orderId: null }),
  book: (b) => api.post<Booking>('/v1/bookings', b),
  cancelBooking: (id) => api.post<Booking>(`/v1/bookings/${enc(id)}/cancel`),
  like: (postId) => api.post<LikeResult>(`/v1/posts/${enc(postId)}/like`),

  createDish: (d) => api.post<Dish>('/v1/admin/dishes', d),
  patchDish: (id, p) => api.patch<Dish>(`/v1/admin/dishes/${enc(id)}`, p),
  deleteDish: (id) => api.delete(`/v1/admin/dishes/${enc(id)}`),
  createCategory: (name) => api.post<Category>('/v1/admin/categories', { name }),
  patchCategory: (id, p) => api.patch<Category>(`/v1/admin/categories/${enc(id)}`, p),
  reorder: async (kind, ids) => {
    await api.put(`/v1/admin/${kind}/order`, { ids });
  },
  createBrand: (b) => api.post<Brand>('/v1/admin/brands', b),
  patchBrand: (id, p) => api.patch<Brand>(`/v1/admin/brands/${enc(id)}`, p),
  deleteBrand: (id) => api.delete(`/v1/admin/brands/${enc(id)}`),
  patchBanner: (id, p) => api.patch(`/v1/admin/banners/${enc(id)}`, p),
  patchSection: (k, p) => api.patch(`/v1/admin/sections/${enc(k)}`, p),
  patchDeal: (p) => api.patch<Deal>('/v1/admin/deal', p),
  patchStore: (p) => api.patch<StoreSettings>('/v1/admin/store', p),
  createPromo: (code, pct) => api.post<Promo>('/v1/admin/promos', { code, pct }),
  patchPromo: (code, on) => api.patch<Promo>(`/v1/admin/promos/${enc(code)}`, { on }),
  createPost: (p) => api.post<Post>('/v1/admin/posts', p),
  patchPost: (id, p) => api.patch<Post>(`/v1/admin/posts/${enc(id)}`, p),
  deletePost: (id) => api.delete(`/v1/admin/posts/${enc(id)}`),

  advanceOrder: (id) => api.post<Order>(`/v1/admin/orders/${enc(id)}/advance`, {}),
  cancelOrder: (id) => api.post<Order>(`/v1/admin/orders/${enc(id)}/cancel`, {}),
  fireTicket: (id) => api.post<Order>(`/v1/admin/orders/${enc(id)}/fire`),
  itemDone: (id, index, done) => api.post<Order>(`/v1/admin/orders/${enc(id)}/items/${index}/done`, { done }),
  setBookingStatus: (id, status) => api.patch<Booking>(`/v1/admin/bookings/${enc(id)}`, { status }),
  replyReview: (id) => api.post<Review>(`/v1/admin/reviews/${enc(id)}/reply`, {}),

  catalogChanged,
  errorMessage: (e) => errorMessage(e),
};

registerRemote(remote);

export type { AdminCatalog };
